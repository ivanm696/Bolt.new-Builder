# ============================================================
# СЛОЙ 1: Классификация утверждений (из Книги 1 — Основы ИИ)
# ============================================================

[STAGE_1_CLASSIFICATION]
# Разбор входного текста на отдельные утверждения

step_1_split = разделить текст на атомарные утверждения
step_2_type  = классифицировать каждое утверждение:
  - fact           = проверяемый факт (дата, число, имя, событие)
  - opinion        = мнение (субъективная оценка)
  - prediction     = предсказание будущего
  - emotional      = эмоциональное высказывание
  - reference      = ссылка на источник
  - contradiction  = внутреннее противоречие

step_3_flag = отметить утверждения, требующие проверки (fact, reference, contradiction)

# ============================================================
# СЛОЙ 2: Формулировка проверочных вопросов (из Книги 2 — Промпт-Инжиниринг)
# ============================================================

[STAGE_2_QUESTIONING]
# Для каждого проверяемого утверждения строится проверочный промпт

techniques:
  - direct_check    = переформулировать утверждение как вопрос и сверить с фактической базой
  - counterfactual  = задать противоположный сценарий и проверить логику
  - detail_probe    = запросить дополнительные детали (лжец обычно не имеет глубины)
  - temporal_check  = проверить хронологию событий
  - source_check    = запросить источник и проверить его существование

# Шаблоны проверочных вопросов:
question_templates:
  fact_q      = "Подтверждается ли, что {утверждение}? Каков источник?"
  contradiction_q = "Совместимо ли утверждение A с утверждением B?"
  detail_q    = "Какие детали подтверждают: {утверждение}?"
  temporal_q  = "Когда произошло: {событие}? Что было до и после?"

# ============================================================
# СЛОЙ 3: Структурный анализ текста (из Книги 3 — Нейросети и Трансформеры)
# ============================================================

[STAGE_3_STRUCTURE_ANALYSIS]
# Паттерны, выявляемые структурным анализом

truth_patterns:
  - логическая связность между утверждениями
  - наличие конкретных деталей (имена, даты, места)
  - согласованность хронологии
  - умеренность формулировок (отсутствие преувеличений)
  - готовность признать незнание

deception_patterns:
  - уклончивые ответы, уклонение от прямого вопроса
  - избыточные обобщения ("все", "никто", "всегда", "никогда")
  - противоречия между ранее сказанным и текущим
  - чрезмерная детализация несущественных моментов (отвлечение)
  - отсутствие деталей на ключевых моментах
  - эмоциональное давление вместо аргументации
  - подмена темы (red herring)
  - апелляция к авторитету без указания источника
  - речевые маркеры неуверенности: "насколько я помню", "кажется", "по-моему" — в ответах на факты

# Веса паттернов (0–10):
weights:
  contradiction           = 9
  evasion                 = 7
  overgeneralization      = 4
  missing_detail_key      = 6
  excessive_detail_minor  = 3
  emotional_pressure      = 5
  topic_switch            = 5
  unsupported_authority   = 4
  uncertainty_on_facts    = 3
  logical_coherence       = -7
  concrete_details        = -5
  chronological_consistency = -6
  admits_unknown          = -3

# ============================================================
# СЛОЙ 4: Прикладной анализ (из Книги 4 — Практическое Применение ИИ)
# ============================================================

[STAGE_4_PRACTICAL_ANALYSIS]
# Прикладные методы поверх структурного анализа

methods:
  - cross_reference    = сверка с внешней фактологической базой
  - consistency_over_time = сравнение утверждений в разные моменты (ловля на изменении версии)
  - comparison_baseline = сравнение с типичными truthful vs deceptive паттернами
  - context_analysis   = учёт контекста: мотивация, обстоятельства, аудитория

scoring:
  # Итоговый score = сумма весов сработавших паттернов
  # Отрицательный score =倾向于 truth
  # Положительный score =倾向于 deception
  # |score| < threshold → inconclusive

# ============================================================
# СЛОЙ 5: Этика, безопасность и границы доверия (из Книги 5 — Этика ИИ)
# ============================================================

[STAGE_5_ETHICS_AND_LIMITS]
# Ограничения системы — что бот НЕ может утверждать категорично

limits:
  - вердикт — вероятностная оценка, не абсолютная истина
  - бот не читает мысли и не анализирует физиологию
  - бот анализирует только текст, не интонацию или видеосигналы
  - ложноположительные результаты возможны при: тревоге, плохой памяти, языковом барьере
  - ложноотрицательные результаты возможны при: подготовленной, продуманной лжи
  - вердикт "inconclusive" обязателен при score ниже порога

safety_rules:
  - никогда не выдавать 100% уверенность
  - всегда указывать matched patterns в выводе
  - при противоречивых сигналах — склоняться к inconclusive
  - не использовать вердикт как единственное основание для решения о человеке

[OUTPUT]
format = structured text
fields:
  1. verdict          = truthful | deceptive | inconclusive
  2. confidence       = 0–95 (никогда не 100)
  3. score            = числовое значение
  4. signals_matched  = список сработавших паттернов с весами
  5. checked_statements = какие утверждения проверялись
  6. notes            = ограничения и контекст данного вердикта

[THRESHOLDS]
confidence_threshold = 60
# Если confidence < 60 → verdict = inconclusive
truth_bias   = -5   # score ниже этого значения → truthful
deception_bias = 5  # score выше этого значения → deceptive
# score между -5 и +5 → inconclusive
